# LogQL — examples for the observability-intermediate course
# Grafana Explore → Loki datasource. Substitute your own label values.

# --- Stream selectors ---
{container="mock-demo-app"}
{container=~"mock-.*"}
{stream="stderr"}

# --- Text filter (line filter) ---
{container="mock-demo-app"} |= "error"
{container="mock-demo-app"} != "health"
{container="mock-demo-app"} |~ "5[0-9]{2}"

# --- JSON / logfmt (if the app logs JSON) ---
# {job="docker"} | json | status >= 500
# {job="docker"} | logfmt | level="error"

# --- Aggregations (metric queries in Loki) ---
sum(rate({container="mock-demo-app"}[5m]))
count_over_time({container="mock-demo-app"}[1h])
bytes_over_time({container="mock-demo-app"}[5m])

# --- Topk by log volume ---
topk(5, sum by (container) (rate({job="docker"}[5m])))
